% LaTeX preamble — Thai language support, generic template
% Used by pandoc with --pdf-engine=xelatex
%
% Document-specific values (title, author, date, etc.) come from
% the YAML frontmatter in each .md file — not hardcoded here.

% --- Font setup (XeLaTeX) ---
\usepackage{fontspec}
\usepackage{polyglossia}
\setmainlanguage{thai}
\setotherlanguage{english}

% Thai font: Sarabun (clean, modern, includes both Thai + Latin glyphs)
\setmainfont{Sarabun}[
  Scale=1.0,
  Ligatures=TeX,
  BoldFont={Sarabun Bold},
  ItalicFont={Sarabun Italic},
  BoldItalicFont={Sarabun Bold Italic}
]
\setsansfont{Sarabun}[
  Scale=1.0,
  Ligatures=TeX,
  BoldFont={Sarabun Bold},
  ItalicFont={Sarabun Italic},
  BoldItalicFont={Sarabun Bold Italic}
]
\setmonofont{Sarabun}[
  Scale=0.85
]

% English font fallback
\newfontfamily\thaifont{Sarabun}
\newfontfamily\englishfont{Sarabun}

% --- Thai line breaking (no space needed between words) ---
\XeTeXlinebreaklocale "th_TH"
\XeTeXlinebreakskip = 0pt plus 1pt minus 0.1pt

% --- Layout ---
\usepackage{geometry}
\geometry{
  a4paper,
  margin=2.5cm,
  top=2.5cm,
  bottom=2.5cm
}

% --- Colors (Minimal Clean: navy blue links, grayscale headings) ---
\usepackage{xcolor}
\definecolor{NavyBlue}{HTML}{1A3A6B}
\definecolor{DarkGray}{HTML}{333333}
\definecolor{LightGray}{HTML}{F5F5F5}
\definecolor{CodeBg}{HTML}{F8F8F8}
\definecolor{TableHead}{HTML}{1A3A6B}

% --- Hyperref ---
% pdftitle, pdfauthor, etc. are auto-set by pandoc from YAML frontmatter
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=NavyBlue,
  urlcolor=NavyBlue,
  citecolor=NavyBlue,
  filecolor=NavyBlue
}

% --- Tables ---
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\renewcommand{\arraystretch}{1.3}

% --- Code blocks ---
\usepackage{fancyvrb}
% Note: 'framed' and 'Shaded' environment are auto-defined by pandoc
% when using --highlight-style. We only set the shade color here.
\definecolor{shadecolor}{HTML}{F8F8F8}

% --- Lists ---
\usepackage{enumitem}
\setlist[itemize]{leftmargin=2em, itemsep=0.3em}
\setlist[enumerate]{leftmargin=2em, itemsep=0.3em}

% --- Section formatting (Minimal Clean) ---
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries\color{NavyBlue}}
  {\filright\Large\chaptertitlename\ \thechapter}
  {0.5ex}
  {\titlerule\vspace{1ex}\filright}
  [\vspace{0.5ex}\titlerule]

% Force section (x.x) to start on a new page, but NOT subsection (x.x.x)
\newcommand{\sectionbreak}{\newpage}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{DarkGray}}
  {\thesection}{1em}{}

\titleformat{\subsection}
  {\normalfont\large\bfseries\color{DarkGray}}
  {\thesubsection}{1em}{}

% --- Header / Footer ---
% Header shows document title and subtitle.
% Values are passed via --variable=title:/subtitle: in build.sh
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\color{DarkGray}\docTitle}
\fancyhead[R]{\small\color{DarkGray}\docSubtitle}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{NavyBlue}\leaders\hrule height \headrulewidth\hfill}}

% --- TOC formatting ---
\usepackage{tocloft}
\renewcommand{\cftchapfont}{\bfseries}
\renewcommand{\cftsecfont}{\normalfont}
\setlength{\cftbeforechapskip}{0.5em}

% --- Image scaling ---
\usepackage{graphicx}
\usepackage{adjustbox}
\usepackage{float}

% Allow figures to float to best position (here, top, bottom, page)
% Center diagrams horizontally on the page
\let\oldfigure\figure
\let\endoldfigure\endfigure
\renewenvironment{figure}[1][]{\oldfigure[!htbp]\centering}{\endoldfigure}

% Constrain image size to fit within page while keeping aspect ratio.
% width=\linewidth makes wide images fill the page; keepaspectratio + height
% limit scales down tall images. \centering (in figure env) centers narrow ones.
\let\oldincludegraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \oldincludegraphics[#1,width=\linewidth,height=0.85\textheight,keepaspectratio]{#2}%
}

% --- Page breaks ---
\newcommand{\newpagebreak}{\newpage}

% --- Document metadata (overridden by temp preamble in build.sh) ---
% Default values; build.sh generates a second preamble that \renewcommand these.
\newcommand{\docTitle}{Document Title}
\newcommand{\docSubtitle}{}
\newcommand{\docAuthor}{}
\newcommand{\docDate}{}
\newcommand{\docType}{}

% --- Custom Title Page (cover) ---
% Values are passed via --variable= in build.sh (read from YAML frontmatter)
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \thispagestyle{empty}
    % Top accent bar
    \noindent
    \colorbox{NavyBlue}{\rule{0pt}{1.2cm}\hspace{\textwidth}\rule{0pt}{1.2cm}}

    \vspace{3cm}

    \begin{center}
      {\fontsize{30pt}{38pt}\selectfont\bfseries\color{NavyBlue}
        \docTitle\par}

      \vspace{0.3cm}

      {\fontsize{18pt}{22pt}\selectfont\color{DarkGray}
        \docSubtitle\par}
    \end{center}

    \vspace{2.5cm}

    \begin{center}
      \begin{tabular}{rl}
        \large\color{DarkGray}
        \textbf{ผู้จัดทำ:}    & \docAuthor \\[0.6em]
        \textbf{วันที่:}      & \docDate \\[0.6em]
        \textbf{เอกสาร:}      & \docType \\
      \end{tabular}
    \end{center}

    \vfill

    % Bottom accent bar
    \noindent
    \colorbox{NavyBlue}{\rule{0pt}{0.4cm}\hspace{\textwidth}\rule{0pt}{0.4cm}}
  \end{titlepage}
}
